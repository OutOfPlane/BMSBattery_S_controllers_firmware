cmake_minimum_required(VERSION 3.10)
project(tinyFOC C)

# Set C standard
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Source files for the main executable
file(GLOB SRCS "Lib/*.c")

# Tests executable
add_executable(main
    main.c
    ${SRCS}
    StdPeriphLib/src/stm8s_itc.c
	StdPeriphLib/src/stm8s_clk.c
	StdPeriphLib/src/stm8s_iwdg.c
	StdPeriphLib/src/stm8s_gpio.c
	StdPeriphLib/src/stm8s_exti.c
	StdPeriphLib/src/stm8s_uart2.c
	StdPeriphLib/src/stm8s_tim1.c
	StdPeriphLib/src/stm8s_tim2.c
	StdPeriphLib/src/stm8s_adc1.c
	StdPeriphLib/src/stm8s_flash.c
)
target_link_libraries(main PRIVATE)
target_include_directories(main PRIVATE
    Lib
    StdPeriphLib/inc
    .
)

target_compile_options(main PRIVATE
    "-mstm8"
    "--std-c99"
    "--nolospre"
)

target_link_options(main PRIVATE
    "-mstm8"
    "--std-c99"
    "--nolospre"
    "--out-fmt-ihx"
)

set(NEW_NAME "tinyfoc.hex")

add_custom_command(
    TARGET main 
    POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy 
            $<TARGET_FILE:main> 
            "${CMAKE_SOURCE_DIR}/${NEW_NAME}"
    COMMENT "Copying binary to project root and renaming to ${NEW_NAME}"
)
